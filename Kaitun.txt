local PlaceID = game.PlaceId
local AllIDs = {}
local IDFileName = "HopIDs.json"
local foundAnything = ""

local function LoadIDs()
    if isfile and isfile(IDFileName) then
        local content = readfile(IDFileName)
        local success, decoded = pcall(function()
            return HttpService:JSONDecode(content)
        end)
        if success and type(decoded) == "table" then
            AllIDs = decoded
        end
    end
end

local function SaveIDs()
    if writefile then
        local content = HttpService:JSONEncode(AllIDs)
        writefile(IDFileName, content)
    end
end

-- Fungsi utama hopping
function HopServer()

    LoadIDs()

    local function TPReturner()
        local url
        if foundAnything == "" then
            url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", PlaceID)
        else
            url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100&cursor=%s", PlaceID, foundAnything)
        end

        local success, response = pcall(function()
            return HttpService:JSONDecode(game:HttpGet(url))
        end)
        if not success then return end

        local Site = response
        if Site.nextPageCursor and Site.nextPageCursor ~= "null" then
            foundAnything = Site.nextPageCursor
        end

        local fallbackServer = nil

        for _, v in pairs(Site.data) do
            if not getgenv().AutoHopEnabled then return end

            local ID = tostring(v.id)
            local possible = true

            -- skip kalau ID pernah dicoba
            for _, existing in ipairs(AllIDs) do
                if ID == existing then
                    possible = false
                    break
                end
            end

            if possible and v.playing < v.maxPlayers then
                if not fallbackServer then
                    fallbackServer = ID
                end

                if v.playing >= 2 then
                    table.insert(AllIDs, ID)
                    SaveIDs()
                    wait()
                    pcall(function()
                        TeleportService:TeleportToPlaceInstance(PlaceID, ID, Players.LocalPlayer)
                    end)
                    wait(0.1)
                    return
                end
            end
        end

        -- Kalau tidak ada yang ≥2, pakai fallback server acak
        if fallbackServer then
            table.insert(AllIDs, fallbackServer)
            SaveIDs()
            wait()
            pcall(function()
                TeleportService:TeleportToPlaceInstance(PlaceID, fallbackServer, Players.LocalPlayer)
            end)
            wait(0.1)
        end
    end

    while getgenv().AutoHopEnabled do
        pcall(TPReturner)
        wait(0.2)
    end
end

-- Fungsi untuk menonaktifkan atau mengurangi efek grafis yang membebani FPS
local function boostFPS()
    -- Menonaktifkan Bayangan Global
    Lighting.GlobalShadows = false

    -- Menurunkan kualitas rendering grafis
    settings().Rendering.QualityLevel = 1  -- Mengatur ke kualitas render terendah    

    -- Menonaktifkan efek partikel dan ledakan
    for _, object in ipairs(Workspace:GetDescendants()) do
        if object:IsA("ParticleEmitter") or object:IsA("Trail") or object:IsA("Smoke") or object:IsA("Fire") then
            object.Enabled = false
        elseif object:IsA("Explosion") then
            object.Visible = false
        end
    end

    -- Mengurangi kualitas material bagian objek
    for _, object in ipairs(Workspace:GetDescendants()) do
        if object:IsA("BasePart") then
            object.Material = Enum.Material.Plastic
            object.Reflectance = 0
        end
    end

    -- Menonaktifkan clothing (pakaian)
    for _, obj in ipairs(Players:GetPlayers()) do
        if obj.Character then
            for _, clothing in ipairs(obj.Character:GetDescendants()) do
                if clothing:IsA("Shirt") or clothing:IsA("Pants") then
                    clothing:Destroy()
                end
            end
        end
    end
end

local lastPos = nil
local lastActivity = tick()

-- Deteksi gerakan pemain lokal
RunService.Heartbeat:Connect(function()
    local char = player.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then
            local pos = hrp.Position
            if lastPos and (pos - lastPos).Magnitude > 0.01 then
                lastActivity = tick()
            end
            lastPos = pos
        end
    end
end)

-- Deteksi idle dan pemain lain → AutoHop
task.spawn(function()
    while task.wait(1) do
        if not getgenv().AutoHopEnabled then continue end

        local idleDuration = tick() - lastActivity
        local char = player.Character
        local myHrp = char and char:FindFirstChild("HumanoidRootPart")
        if not myHrp then continue end

        -- Cek pemain lain di workspace.Characters
        local nearPlayer = false
        local charFolder = workspace:FindFirstChild("Characters")
        if charFolder then
            for _, other in ipairs(Players:GetPlayers()) do
                if other ~= player then
                    local otherChar = charFolder:FindFirstChild(other.Name)
                    local hrp = otherChar and otherChar:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        local dist = (myHrp.Position - hrp.Position).Magnitude
                        if dist <= 50 then
                            nearPlayer = true
                            break
                        end
                    end
                end
            end
        end

        -- Jika idle terlalu lama atau ada pemain dekat
        if idleDuration >= 60 or nearPlayer then
            if type(HopServer) == "function" then
                HopServer()
            end
            break -- stop loop agar tidak spam
        end
    end
end)

local function autoRejoin()
    local gameId = game.PlaceId
    
    local function rejoin()
        task.spawn(function()
            local success, err = pcall(function()
                TeleportService:Teleport(gameId, Players.LocalPlayer)
            end)
            
            if not success then
                warn("Gagal rejoin:", err)
                task.wait(5)
                rejoin()
            end
        end)
    end

    Players.PlayerRemoving:Connect(function(player)
        if player == Players.LocalPlayer then
            warn("Player dikeluarkan, mencoba rejoin...")
            task.wait(2)
            rejoin()
        end
    end)

    TeleportService.TeleportFailed:Connect(function(player, teleportResult, errorMessage)
        if player == Players.LocalPlayer then
            warn("Teleport gagal:", teleportResult, errorMessage)
            task.wait(2)
            rejoin()
        end
    end)
end

local function tambahStatOtomatis()
    while true do
        local stats = player:FindFirstChild("Data") and player.Data:FindFirstChild("Stats")
        if not stats then return end

        local melee = stats:FindFirstChild("Melee")
        local defense = stats:FindFirstChild("Defense")
        local sword = stats:FindFirstChild("Sword")
        
        if not (melee and defense and sword) then return end

        local meleeLevel = melee:FindFirstChild("Level")
        local defenseLevel = defense:FindFirstChild("Level")
        local swordLevel = sword:FindFirstChild("Level")

        if not (meleeLevel and defenseLevel and swordLevel) then return end


        if meleeLevel.Value < 2650 then
            if defenseLevel.Value < meleeLevel.Value - 500 then
                ReplicatedStorage.Remotes.CommF_:InvokeServer("AddPoint", "Defense", 100)
            else
                ReplicatedStorage.Remotes.CommF_:InvokeServer("AddPoint", "Melee", 500)
            end
        
        -- Jika melee sudah mencapai 2650, lanjutkan menambah defense dengan kelipatan 100
        elseif defenseLevel.Value < 2650 then
            ReplicatedStorage.Remotes.CommF_:InvokeServer("AddPoint", "Defense", 100)

        -- Jika defense sudah mencapai 2650, lanjutkan menambah sword dengan kelipatan 100
        elseif swordLevel.Value < 2650 then
            ReplicatedStorage.Remotes.CommF_:InvokeServer("AddPoint", "Sword", 100)
        end

        task.wait(0.5)  -- Interval waktu antara penambahan stat
    end
end

local function GetNearestPortal(pos)
    if not pos or #PortalPos == 0 then return nil end
    local shortest = math.huge
    local closest = nil
    for _, portal in pairs(PortalPos) do
        local dist = (portal - pos).Magnitude
        if dist < shortest then
            shortest = dist
            closest = portal
        end
    end
    return closest
end

local function tweenTo(targetCFrame)
    stopTween()

    local targetPos = targetCFrame.Position
    local portal = GetNearestPortal(targetPos)
    local distToTarget = (block.Position - targetPos).Magnitude
    local distPortalToTarget = portal and (portal - targetPos).Magnitude or math.huge

    if distToTarget > distPortalToTarget + 250 then
        -- Teleport block dan player ke portal
        block.CFrame = CFrame.new(portal)
        if Player.Character and Player.Character.PrimaryPart then
            Player.Character.PrimaryPart.CFrame = block.CFrame
        end
        task.wait(3)
    end

    -- Hitung durasi tween
    local horizontalDist = (Vector3.new(targetPos.X, 0, targetPos.Z) - Vector3.new(block.Position.X, 0, block.Position.Z)).Magnitude
    if horizontalDist < 1 then return end

    local speed = getgenv().TweenSpeed or 325
    local duration = horizontalDist / speed

    -- Tween block ke target
    currentTween = TweenService:Create(block,
        TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out),
        {CFrame = targetCFrame}
    )
    currentTween:Play()

    -- Player nempel ke block selama tween
    task.spawn(function()
        while currentTween and currentTween.PlaybackState == Enum.PlaybackState.Playing do
            if Player.Character and Player.Character.PrimaryPart then
                Player.Character.PrimaryPart.CFrame = block.CFrame
            end
            task.wait()
        end
    end)

    currentTween.Completed:Wait()

    -- Reset humanoid biar normal
    if humanoid then
        humanoid.AutoRotate = true
        humanoid.WalkSpeed = 16
    end
end

function tweenToRaid(cframe)
    stopTween()

    local targetPos = cframe.Position
    local stableCFrame = CFrame.new(targetPos)
    getgenv().LastTweenTarget = stableCFrame

    -- Hitung jarak horizontal (tanpa Y)
    local horizontalDist = (Vector3.new(targetPos.X, 0, targetPos.Z) - Vector3.new(hrp.Position.X, 0, hrp.Position.Z)).Magnitude
    if horizontalDist < 1 then return end

    local speed = getgenv().TweenSpeed or 325
    local duration = horizontalDist / speed
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)

    -- Tween langsung ke HRP
    currentTween = TweenService:Create(hrp, tweenInfo, {CFrame = stableCFrame})
    currentTween:Play()

    currentTween.Completed:Wait()

    -- Kembalikan kontrol karakter
    if humanoid then
        humanoid.AutoRotate = true
        humanoid.WalkSpeed = 16
    end
end

-- Utils
local function WaitForChild(parent, childName, timeout)
    timeout = timeout or 5
    local start = os.clock()
    local child = parent:FindFirstChild(childName)

    while not child and (os.clock() - start) < timeout do
        task.wait(0.1)
        child = parent:FindFirstChild(childName)
    end

    return child
end

local function VerifyTool(ToolName)
    local plrChar = player.Character or player.CharacterAdded:Wait()
    local plrBag = player:WaitForChild("Backpack")
    return plrChar:FindFirstChild(ToolName) or plrBag:FindFirstChild(ToolName)
end

local function GetToolLevel(ToolName)
    task.wait(0.1)
    local Tool = VerifyTool(ToolName)
    if Tool and Tool:FindFirstChild("Level") then
        return Tool.Level.Value
    end
    return 0
end

local function EquipTool(ToolName)
    while not player.Character do
        task.wait(0.5)
    end

    local character = player.Character
    if character:FindFirstChild(ToolName) then
        return true
    end

    local backpack = player:FindFirstChild("Backpack")
    if not backpack then return false end

    local tool = WaitForChild(backpack, ToolName, 5)
    if not tool then return false end

    tool.Parent = character
    local equippedTool = WaitForChild(character, ToolName, 5)
    if not equippedTool then return false end

    local equipEvent = equippedTool:FindFirstChild("EquipEvent")
    if equipEvent and equipEvent:IsA("RemoteEvent") then
        equipEvent:FireServer(true)
        return true
    end

    return false
end

local function BuyAndEquip(RemoteName, ToolName)
    local flagName = "alreadyBought"..ToolName:gsub(" ", "")
    local maxWait = 15

    if not VerifyTool(ToolName) then
        if not _G.flags[flagName] then
            ReplicatedStorage.Remotes.CommF_:InvokeServer(RemoteName)
            _G.flags[flagName] = true
        end

        local start = os.time()
        while not VerifyTool(ToolName) and (os.time() - start) < maxWait do
            task.wait(1)
        end
    end

    for i = 1, 3 do
        if EquipTool(ToolName) then
            return true
        end
        task.wait(1)
    end

    return false
end

function autoV()
    -- Tunggu Players service siap
    local Players = game:WaitForChild("Players")

    -- Tunggu workspace.Characters siap
    local workspaceCharacters = workspace:WaitForChild("Characters")

    -- Ambil nama player
    local playerName = LocalPlayer.Name

    while true do
        -- Cek apakah ada Black Leg atau Death Step
        local char = workspaceCharacters:FindFirstChild(playerName)
        if char then
            local hasBlackLeg = char:FindFirstChild("Black Leg")
            local hasDeathStep = char:FindFirstChild("Death Step")
            
            if hasBlackLeg or hasDeathStep then
                print("Fighting style ditemukan, mulai auto skill V.")
                while true do
                    -- Pastikan char masih ada
                    char = workspaceCharacters:FindFirstChild(playerName)
                    if not char then
                        break
                    end
                    -- Simulate key press V
                    game:GetService("VirtualInputManager"):SendKeyEvent(true, "V", false, game)
                    wait(0.1)
                    game:GetService("VirtualInputManager"):SendKeyEvent(false, "V", false, game)
                    -- Tunggu 30 detik
                    wait(5)
                    
                    -- Cek lagi apakah fighting style masih ada
                    char = workspaceCharacters:FindFirstChild(playerName)
                    if not char or (not char:FindFirstChild("Black Leg") and not char:FindFirstChild("Death Step")) then
                        print("Fighting style hilang, berhenti spam skill.")
                        break
                    end
                end
            else
                print("Tidak ada fighting style, tunggu 5 menit sebelum cek lagi.")
                wait(10) -- 5 menit
            end
        else
            print("Karakter tidak ditemukan, tunggu 5 menit sebelum cek lagi.")
            wait(10)
        end
        wait(1) -- Tunggu sebentar sebelum loop cek ulang
    end
end

local function CanStoreFruit(fruitName)

    if getgenv().CanStoreFruit ~= nil then
        return getgenv().CanStoreFruit
    end

    if not isSecondSea then return true end

        local character = Player.Character

        local function handleContainer(container)
            for _, item in ipairs(container:GetChildren()) do
                if item:IsA("Tool") and item:FindFirstChild("Fruit") then
                    local fruitName = item.Name
                    if CanStoreFruit(fruitName) then
                        local fruitCode = Get_Fruit(fruitName)
                        if fruitCode then
                            Remote:InvokeServer("StoreFruit", fruitCode, item)
                        end
                    end
                end
            end
        end

        if character then handleContainer(character) end
        if backpack then handleContainer(backpack) end

        task.wait(60)
    end
end)

local function SelesaikanBartiloQuest()
    local function ambilProgress()
        local ok, progress = pcall(function()
            return ReplicatedStorage.Remotes.CommF_:InvokeServer("BartiloQuestProgress")
        end)
        return ok and progress or nil
    end

    local progress = ambilProgress()
    if not progress then
        warn("Gagal ambil progress Bartilo!")
        return
    end
    if progress.DidPlates then
        print("Semua Bartilo Quest sudah selesai!")
        progress.SudahSelesaiBartilo = true
        saveProgress()
        return
    end

    -- STEP 1: Bunuh Bandit
    if not progress.KilledBandits then
        ReplicatedStorage.Remotes.CommF_:InvokeServer("StartQuest", "BartiloQuest", 1)
        task.wait(1)
        local swanSpawns = spawnPositions["Swan Pirate"]
        while true do
            progress = ambilProgress()
            if progress.KilledBandits then break end
            local found = false
            for _, enemy in pairs(workspace.Enemies:GetChildren()) do
                if enemy.Name == "Swan Pirate" and enemy:FindFirstChild("Humanoid") and enemy.Humanoid.Health > 0 then
                    serangMusuh(enemy)
                    found = true
                    break
                end
            end
            if not found and swanSpawns then
                tweenTo(swanSpawns[math.random(1, #swanSpawns)])
                task.wait(0.2)
            end
        end
        print("Step Bandit selesai!")
    end

    -- STEP 2: Bunuh Jeremy
    progress = ambilProgress()
    if not progress.KilledSpring then
        local jeremySpawns = spawnPositions["Jeremy"]
        if jeremySpawns and #jeremySpawns > 0 then
            tweenTo(jeremySpawns[1])
            task.wait(0.1)
        end

        while true do
            progress = ambilProgress()
            if progress.KilledSpring then break end
            local foundJeremy = false
            for _, enemy in pairs(workspace.Enemies:GetChildren()) do
                if enemy.Name == "Jeremy" and enemy:FindFirstChild("Humanoid") and enemy.Humanoid.Health > 0 then
                    foundJeremy = true
                    serangMusuh(enemy)
                    break
                end
            end
            if not foundJeremy then
                getgenv().AutoHopEnabled = true
                HopServer()
                return
            end
            task.wait(0.5)
        end
        print("Step Jeremy selesai!")
    end

    -- STEP 3: Plates, validasi per-plate sampai benar-benar aktif
    progress = ambilProgress()
    if not progress.DidPlates then
        local plateFolder = workspace.Map.Dressrosa:FindFirstChild("BartiloPlates")
        if plateFolder then
            for i = 1, 8 do
                local plate = plateFolder:FindFirstChild("Plate" .. i)
                if plate then
                    repeat
                        teleport(plate.CFrame)
                        task.wait(2)
                    until plate.Color == Color3.fromRGB(150, 255, 125)
                end
            end
        end
        -- Pastikan DidPlates dari server sudah TRUE
        while true do
            progress = ambilProgress()
            if progress.DidPlates then break end
            task.wait(0.5)
        end
        print("Plates selesai!")
    end

    progress.SudahSelesaiBartilo = true
    saveProgress()
    print("Semua Bartilo Quest selesai!")
end

local function jalankanEliteHunter()
    local player = game.Players.LocalPlayer

    local CommF = ReplicatedStorage.Remotes.CommF_
    local guideFrame = player.PlayerGui.Main.Guide.LeftFrame.IconFrame.ViewportFrame

    if (progress.JumlahEliteHunter or 0) >= 30 then
        -- Posisi pedang Yama
        local targetPos = Vector3.new(5251.9, 17.18, 453.6)
        local clickDetector = workspace.Map.Waterfall.SealedKatana.Hitbox.ClickDetector

        -- Tween ke posisi
        tweenTo(CFrame.new(targetPos))
        task.wait(1)

        -- Klik pedang 10 kali
        for i = 1, 10 do
            clickDetector:FireClick()
            task.wait(0.3)
        end

        -- Tandai progress sudah ambil Yama
        progress.YamaSudahDiambil = true
        saveProgress()

        return
    end

    -- Mulai quest Elite Hunter
    CommF:InvokeServer("EliteHunter")
    task.wait(0.5)

    local bossName = nil
    for _, nama in ipairs({ "Diablo", "Deandre", "Urban" }) do
        if guideFrame:FindFirstChild(nama) then
            bossName = nama
            break
        end
    end

    if not bossName then
        return HopServer()
    end

    -- Ambil boss
    local bossModel = ReplicatedStorage:FindFirstChild(bossName)
    local hrp = bossModel and bossModel:FindFirstChild("HumanoidRootPart")

    if not hrp then
        local bossEnemy = workspace:FindFirstChild("Enemies") and workspace.Enemies:FindFirstChild(bossName)
        if bossEnemy and bossEnemy:FindFirstChild("HumanoidRootPart") then
            hrp = bossEnemy.HumanoidRootPart
        end
    end

    if not hrp then
        return
    end

    tweenTo(hrp.CFrame)
    task.wait(1)

    -- Cari musuh hidup
    for _, musuh in ipairs(workspace.Enemies:GetChildren()) do
        if musuh.Name == bossName and musuh:FindFirstChild("Humanoid") and musuh.Humanoid.Health > 0 then
            
            -- Pasang event listener
            musuh.Humanoid.Died:Connect(function()
                progress.JumlahEliteHunter = (progress.JumlahEliteHunter or 0) + 1
                saveProgress()
            end)

            -- Serang musuh
            serangMusuh(musuh)
            return
        end
    end
end

    end

    -- Mulai pembelian bones
    local bonesSebelum = getJumlahBones()
    beliSekali()
    local bonesSesudah = getJumlahBones()

    if bonesSebelum - bonesSesudah >= 1 then

        if bonesSesudah < 800 then
            ambilQuest(hauntedcastle2QuestCFrame, "HauntedQuest2", 2)

            cariDanSerangMusuh("Posessed Mummy")

            return
        end
        jalankanEliteHunter()

        jalankanEliteHunter()
    end

local function attackEnemy(enemyName)
        while true do
            local enemy = nil
            for _, mob in pairs(workspace:WaitForChild("Enemies"):GetChildren()) do
                if mob.Name == enemyName and mob:FindFirstChild("Humanoid") and mob.Humanoid.Health > 0 then
                    enemy = mob
                    break
                end
            end
            
            -- Jika musuh tidak ditemukan, keluar loop
            if not enemy then break end
            
            -- Cari bagian Head musuh dan pastikan karakter pemain valid
            local enemyHead = enemy:FindFirstChild("Head")
            local humanoid = enemy:FindFirstChild("Humanoid")
            
            if enemyHead and humanoid and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                -- Posisikan pemain di atas musuh
                local hrp = player.Character.HumanoidRootPart
                local newPos = enemyHead.Position + Vector3.new(0, 30, 0)
                hrp.CFrame = CFrame.new(newPos) * CFrame.Angles(0, hrp.CFrame.Rotation.Y, 0)
                
                -- Serang musuh
                ReplicatedStorage.Modules.Net["RE/RegisterAttack"]:FireServer(0.5)
                _G.SendHitsToServer(enemyHead, {})
            end
            
            task.wait(0.01)
        end
        
        return true
    end

-- Cek apakah buah sudah ada di backpack
local function hasFruitInBackpack(fruitName)
    local backpack = player:FindFirstChild("Backpack")
    if not backpack then return false end
    for _, tool in pairs(backpack:GetChildren()) do
        if tool:IsA("Tool") and tool.Name == fruitName then
            return true
        end
    end
    return false
end

-- Coba unstore semua buah, stop jika ada satu yang berhasil
local function TryUnstoreCommonFruitOnce()
    if not isThirdSea then return end

    local CommF = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("CommF_")

    for fruitName, fruitCode in pairs(fruitUnstoreNameMap) do
        if hasFruitInBackpack(fruitName) then
            return
        end

        local success = pcall(function()
            CommF:InvokeServer("LoadFruit", fruitCode)
        end)

        task.wait(0.5)

        if hasFruitInBackpack(fruitName) then
            return
        end
    end
end
